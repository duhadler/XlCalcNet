using System;
using System.Numerics;
using FixedPrecNet;

namespace EvaluateCS
{
    public class Program
    {
        public double[,] test4()
        {

            int numPoints = 256;
            int ExtraDt = 3;
            double tmin = 0;
            double tmax = 6.28318530717959;
            double[,] d3 = new double[3, numPoints+ExtraDt+1];
            double t = tmin;
            double dt = (tmax - tmin) / (numPoints);
            for (int i = 0; i < numPoints +ExtraDt; i++)
            {
                // Start function definition
var r = Math.Sqrt(3) / 3;
var x = Math.Cos(t) - 0.5;
var y = Math.Sin(t) - r / 2;
var z = Math.Cos(3 * t) / 3;
                // End function definition
                d3[0, i] = x;;
                d3[1, i] = y;
                d3[2, i] = z;
                t += dt;
            }
            return d3;
        }
    }
}
